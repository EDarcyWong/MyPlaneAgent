## 架构总览

### 5 层架构图

```
┌─────────────────────────────────────────────────┐
│                  Agent Core                     │  Layer 1
│       Planner / Executor / Replan / Memory      │
└────────────────────┬────────────────────────────┘
                     │ 请求能力
┌────────────────────▼────────────────────────────┐
│            Capability Registry                  │  Layer 2
│  browser / git / file / coding / office / db    │
│  注册表：能力元数据 + 执行入口                     │
└────────────────────┬────────────────────────────┘
                     │ 调度 Skill
┌────────────────────▼────────────────────────────┐
│              Skill Platform                     │  Layer 3
│    Tool / Workflow / Prompt / Resource          │
│    统一封装：参数验证 / 权限检查 / 日志记录        │
└────────────────────┬────────────────────────────┘
                     │ 提交任务
┌────────────────────▼────────────────────────────┐
│         Python Runtime Manager                  │  Layer 4
│       venv / Worker Pool / IPC / Sandbox        │
│       进程管理 / 依赖隔离 / 通信协议              │
└────────────────────┬────────────────────────────┘
                     │ 执行
              ┌──────┴──────┐
              ▼             ▼
        MCP Adapter    Native Python               Layer 5
        标准互操作      本地高性能
              │             │
        External MCP   直接执行
```

### 数据流向

```
用户请求
    ↓
Agent Core (规划任务)
    ↓
Capability Registry (查找能力 browser.screenshot)
    ↓
Skill Platform (封装调用)
    ↓
Python Runtime Manager (分配 Worker)
    ↓
    ├─→ Native Python (本地执行)
    └─→ MCP Adapter (远程 MCP 服务)
    ↓
返回结果 → Agent Core → 用户
```

## Layer 1: Agent Core

### 职责

1. **Planner**: 任务规划，将用户意图分解为能力调用序列
2. **Executor**: 执行规划，调用 Capability Registry
3. **Replan**: 执行失败时重新规划
4. **Memory**: 短期记忆（对话上下文）+ 长期记忆（知识库）

### 实现

**文件**: `electron/main/agent/core/agent-core.ts`

```typescript
import type { CapabilityRegistry } from './capability-registry.js'
import type { AgentMemory } from './memory.js'

export type AgentTask = {
  id: string
  description: string
  context: {
    workspace: string
    files?: string[]
    userIntent: string
  }
}

export type AgentPlan = {
  taskId: string
  steps: Array<{
    capability: string        // 能力名称：browser.screenshot
    args: Record<string, unknown>
    dependsOn?: number[]      // 依赖的步骤索引
  }>
  reasoning: string           // 规划理由
}

export type ExecutionResult = {
  success: boolean
  outputs: unknown[]
  errors?: string[]
  elapsedMs: number
}

export class AgentCore {
  constructor(
    private capabilities: CapabilityRegistry,
    private memory: AgentMemory,
    private modelClient: ModelClient
  ) {}
  
  // 主流程：规划 → 执行 → (失败则重新规划)
  async run(task: AgentTask, signal: AbortSignal): Promise<ExecutionResult> {
    let plan = await this.plan(task)
    let attempt = 0
    const maxAttempts = 3
    
    while (attempt < maxAttempts) {
      signal.throwIfAborted()
      
      const result = await this.execute(plan, signal)
      
      if (result.success) {
        // 成功：存储记忆
        await this.memory.store({
          taskId: task.id,
          plan,
          result,
          timestamp: Date.now()
        })
        return result
      }
      
      // 失败：重新规划
      attempt++
      if (attempt < maxAttempts) {
        plan = await this.replan(task, plan, result)
      }
    }
    
    throw new Error(`Task failed after ${maxAttempts} attempts`)
  }
  
  // Planner: 调用大模型生成计划
  async plan(task: AgentTask): Promise<AgentPlan> {
    // 获取可用能力列表
    const capabilities = this.capabilities.list()
    
    // 构造 Prompt
    const prompt = `
You are a task planner. Given a user task, break it down into capability calls.

Available capabilities:
${capabilities.map(c => `- ${c.name}: ${c.description}`).join('\n')}

User task: ${task.description}
Context: ${JSON.stringify(task.context)}

Output a JSON plan with steps:
{
  "reasoning": "why this plan works",
  "steps": [
    { "capability": "file.read", "args": { "path": "..." } },
    { "capability": "browser.screenshot", "args": { "url": "..." } }
  ]
}
`
    
    const response = await this.modelClient.complete({
      messages: [{ role: 'user', content: prompt }],
      temperature: 0.2
    })
    
    const plan = JSON.parse(response.content)
    
    return {
      taskId: task.id,
      steps: plan.steps,
      reasoning: plan.reasoning
    }
  }
  
  // Executor: 按顺序执行步骤
  async execute(plan: AgentPlan, signal: AbortSignal): Promise<ExecutionResult> {
    const startTime = Date.now()
    const outputs: unknown[] = []
    const errors: string[] = []
    
    for (const [index, step] of plan.steps.entries()) {
      signal.throwIfAborted()
      
      try {
        // 检查依赖
        if (step.dependsOn) {
          for (const dep of step.dependsOn) {
            if (errors[dep]) {
              throw new Error(`Dependency step ${dep} failed`)
            }
          }
        }
        
        // 调用能力
        const result = await this.capabilities.execute(
          step.capability,
          step.args,
          signal
        )
        
        outputs[index] = result
      } catch (error) {
        errors[index] = error instanceof Error ? error.message : String(error)
      }
    }
    
    return {
      success: errors.length === 0,
      outputs,
      errors: errors.length > 0 ? errors : undefined,
      elapsedMs: Date.now() - startTime
    }
  }
  
  // Replan: 失败后重新规划
  async replan(
    task: AgentTask,
    previousPlan: AgentPlan,
    previousResult: ExecutionResult
  ): Promise<AgentPlan> {
    const prompt = `
The previous plan failed. Generate a new plan.

Previous plan:
${JSON.stringify(previousPlan, null, 2)}

Errors:
${previousResult.errors?.join('\n')}

Generate an alternative plan that avoids these errors.
`
    
    const response = await this.modelClient.complete({
      messages: [{ role: 'user', content: prompt }],
      temperature: 0.3
    })
    
    const newPlan = JSON.parse(response.content)
    
    return {
      taskId: task.id,
      steps: newPlan.steps,
      reasoning: newPlan.reasoning
    }
  }
}
```

## Layer 2: Capability Registry

### 职责

1. **注册能力**: Skill 启动时自动注册能力（browser.screenshot, git.commit）
2. **查询能力**: Agent Core 查询可用能力
3. **路由执行**: 根据能力来源（Python Native / MCP）路由到正确的执行层

### 实现

**文件**: `electron/main/agent/core/capability-registry.ts`

```typescript
export type Capability = {
  name: string                    // browser.screenshot
  category: string                // browser
  description: string
  parameters: Record<string, unknown>  // JSON Schema
  
  // 能力来源
  source: {
    type: 'skill' | 'mcp'
    skillId?: string              // file-operations
    mcpServer?: string            // puppeteer-mcp
  }
  
  // 执行路径
  runtime: 'python-native' | 'mcp'
  
  // 权限要求
  permissions: string[]
}

export class CapabilityRegistry {
  private capabilities = new Map<string, Capability>()
  
  constructor(
    private skillPlatform: SkillPlatform,
    private mcpAdapter: McpAdapter
  ) {}
  
  // 注册能力
  register(capability: Capability): void {
    if (this.capabilities.has(capability.name)) {
      throw new Error(`Capability already registered: ${capability.name}`)
    }
    
    this.capabilities.set(capability.name, capability)
    console.log(`[Registry] Registered: ${capability.name}`)
  }
  
  // 取消注册
  unregister(name: string): void {
    this.capabilities.delete(name)
  }
  
  // 查询能力
  get(name: string): Capability | undefined {
    return this.capabilities.get(name)
  }
  
  // 列出所有能力
  list(category?: string): Capability[] {
    const caps = Array.from(this.capabilities.values())
    
    if (category) {
      return caps.filter(c => c.category === category)
    }
    
    return caps
  }
  
  // 按类别分组
  groupByCategory(): Record<string, Capability[]> {
    const groups: Record<string, Capability[]> = {}
    
    for (const cap of this.capabilities.values()) {
      if (!groups[cap.category]) {
        groups[cap.category] = []
      }
      groups[cap.category].push(cap)
    }
    
    return groups
  }
  
  // 执行能力
  async execute(
    name: string,
    args: Record<string, unknown>,
    signal: AbortSignal
  ): Promise<unknown> {
    const capability = this.capabilities.get(name)
    
    if (!capability) {
      throw new Error(`Capability not found: ${name}`)
    }
    
    // 根据运行时类型路由
    if (capability.runtime === 'python-native') {
      // 通过 Skill Platform 执行
      return await this.skillPlatform.executeTool(
        capability.source.skillId!,
        name.split('.')[1],  // browser.screenshot → screenshot
        args,
        signal
      )
    } else if (capability.runtime === 'mcp') {
      // 通过 MCP Adapter 执行
      return await this.mcpAdapter.callMcp(
        capability.source.mcpServer!,
        name,
        args,
        signal
      )
    } else {
      throw new Error(`Unknown runtime: ${capability.runtime}`)
    }
  }
  
  // 从 Skill Platform 自动注册
  async loadFromSkills(): Promise<void> {
    const skills = this.skillPlatform.listSkills()
    
    for (const skill of skills) {
      const tools = skill.capabilities.tools || []
      
      for (const tool of tools) {
        this.register({
          name: `${skill.category}.${tool.name}`,
          category: skill.category,
          description: tool.description,
          parameters: tool.parameters,
          source: {
            type: 'skill',
            skillId: skill.id
          },
          runtime: 'python-native',
          permissions: skill.permissions
        })
      }
    }
  }
  
  // 从 MCP 自动注册
  async loadFromMcp(): Promise<void> {
    const servers = this.mcpAdapter.listServers()
    
    for (const server of servers) {
      const tools = await this.mcpAdapter.listTools(server.id)
      
      for (const tool of tools) {
        this.register({
          name: tool.name,
          category: this.extractCategory(tool.name),
          description: tool.description,
          parameters: tool.parameters,
          source: {
            type: 'mcp',
            mcpServer: server.id
          },
          runtime: 'mcp',
          permissions: []
        })
      }
    }
  }
  
  private extractCategory(name: string): string {
    // browser_screenshot → browser
    // git_commit → git
    const parts = name.split('_')
    return parts[0]
  }
}
```

## Layer 3: Skill Platform

### 职责

1. **Skill 生命周期管理**: 安装、卸载、启用、禁用
2. **四种能力封装**:
   - **Tool**: 单次执行的工具（read_file, screenshot）
   - **Workflow**: 多步骤编排（code_review, refactor）
   - **Prompt**: 提示词模板（用于 Agent 规划）
   - **Resource**: 静态资源（图标、配置）
3. **统一接口**: 参数验证、权限检查、日志记录

**文件**: `electron/main/agent/core/skill-platform.ts`

```typescript
export type SkillManifest = {
  id: string
  name: string
  version: string
  category: string
  description: string
  runtime: {
    type: 'python' | 'nodejs'
    version: string
    entry: string
    venv: boolean
  }
  capabilities: {
    tools?: ToolDefinition[]
    workflows?: WorkflowDefinition[]
    prompts?: PromptDefinition[]
    resources?: Record<string, string>
  }
  permissions: {
    fileSystem?: { read?: string[]; write?: string[] }
    network?: boolean
    process?: boolean
  }
  dependencies: {
    python?: string[]
    npm?: Record<string, string>
    skills?: string[]
    mcp?: { required?: string[]; optional?: string[] }
  }
}

export type SkillInstance = {
  manifest: SkillManifest
  path: string
  enabled: boolean
  installedAt: string
  status: 'idle' | 'loading' | 'ready' | 'error'
  error?: string
}

export class SkillPlatform {
  private skills = new Map<string, SkillInstance>()
  
  constructor(
    private skillsDir: string,
    private runtimeManager: PythonRuntimeManager,
    private capabilityRegistry: CapabilityRegistry
  ) {}
  
  // 初始化：扫描并加载 Skills
  async initialize(): Promise<void> {
    const entries = fs.readdirSync(this.skillsDir, { withFileTypes: true })
    
    for (const entry of entries) {
      if (!entry.isDirectory()) continue
      
      const skillPath = path.join(this.skillsDir, entry.name)
      const manifestPath = path.join(skillPath, 'skill.json')
      
      if (!fs.existsSync(manifestPath)) continue
      
      try {
        const manifest: SkillManifest = JSON.parse(
          fs.readFileSync(manifestPath, 'utf8')
        )
        
        const instance: SkillInstance = {
          manifest,
          path: skillPath,
          enabled: true,
          installedAt: fs.statSync(skillPath).birthtime.toISOString(),
          status: 'idle'
        }
        
        this.skills.set(manifest.id, instance)
        
        // 注册到 Capability Registry
        await this.registerCapabilities(instance)
        
      } catch (error) {
        console.error(`Failed to load skill ${entry.name}:`, error)
      }
    }
  }
  
  // 注册 Skill 的能力到 Registry
  private async registerCapabilities(skill: SkillInstance): Promise<void> {
    if (!skill.enabled) return
    
    const tools = skill.manifest.capabilities.tools || []
    
    for (const tool of tools) {
      this.capabilityRegistry.register({
        name: `${skill.manifest.category}.${tool.name}`,
        category: skill.manifest.category,
        description: tool.description,
        parameters: tool.parameters,
        source: {
          type: 'skill',
          skillId: skill.manifest.id
        },
        runtime: 'python-native',
        permissions: Object.keys(skill.manifest.permissions)
      })
    }
  }
  
  // 执行 Tool
  async executeTool(
    skillId: string,
    toolName: string,
    args: Record<string, unknown>,
    signal: AbortSignal
  ): Promise<unknown> {
    const skill = this.skills.get(skillId)
    
    if (!skill) {
      throw new Error(`Skill not found: ${skillId}`)
    }
    
    if (!skill.enabled) {
      throw new Error(`Skill disabled: ${skillId}`)
    }
    
    // 参数验证
    const tool = skill.manifest.capabilities.tools?.find(t => t.name === toolName)
    if (!tool) {
      throw new Error(`Tool not found: ${toolName}`)
    }
    
    // TODO: 使用 ajv 验证参数
    
    // 权限检查
    await this.checkPermissions(skill, args)
    
    // 提交到 Runtime Manager
    const result = await this.runtimeManager.execute({
      skillId: skill.manifest.id,
      skillPath: skill.path,
      tool: toolName,
      args,
      venv: skill.manifest.runtime.venv
    }, signal)
    
    return result.output
  }
  
  // 权限检查
  private async checkPermissions(
    skill: SkillInstance,
    args: Record<string, unknown>
  ): Promise<void> {
    const perms = skill.manifest.permissions
    
    // 检查文件系统权限
    if (perms.fileSystem) {
      const path = args.path as string | undefined
      if (path) {
        // TODO: 使用 pathspec 检查路径是否在允许范围内
      }
    }
    
    // 检查网络权限
    if (!perms.network && args.url) {
      throw new Error(`Skill ${skill.manifest.id} not permitted to access network`)
    }
  }
  
  // 安装 Skill
  async install(source: string): Promise<SkillInstance> {
    // TODO: 从 Git / npm / 本地路径安装
    throw new Error('Not implemented')
  }
  
  // 卸载 Skill
  async uninstall(skillId: string): Promise<void> {
    const skill = this.skills.get(skillId)
    if (!skill) throw new Error(`Skill not found: ${skillId}`)
    
    // 从 Registry 取消注册
    const tools = skill.manifest.capabilities.tools || []
    for (const tool of tools) {
      this.capabilityRegistry.unregister(`${skill.manifest.category}.${tool.name}`)
    }
    
    // 删除文件
    fs.rmSync(skill.path, { recursive: true, force: true })
    
    this.skills.delete(skillId)
  }
  
  // 列出 Skills
  listSkills(): SkillInstance[] {
    return Array.from(this.skills.values())
  }
}
```

## Layer 4: Python Runtime Manager

### 职责

1. **venv 隔离**: 每个 Skill 独立虚拟环境
2. **Worker Pool**: 进程池管理（最多 8 个并发）
3. **IPC 通信**: JSON-RPC over stdin/stdout
4. **Sandbox**: 可选的 Docker/Podman 沙箱

### 实现

**文件**: `electron/main/agent/core/python-runtime-manager.ts`

```typescript
import { spawn, type ChildProcessWithoutNullStreams } from 'node:child_process'
import { randomUUID } from 'node:crypto'

export type ExecutionRequest = {
  skillId: string
  skillPath: string
  tool: string
  args: Record<string, unknown>
  venv: boolean
  workspace?: string
}

export type ExecutionResponse = {
  output: unknown
  elapsedMs: number
  mcpCalls?: Array<{
    server: string
    tool: string
    result: unknown
  }>
}

type Worker = {
  skillId: string
  process: ChildProcessWithoutNullStreams
  venvPath: string | null
  pending: Map<string, {
    resolve: (value: ExecutionResponse) => void
    reject: (error: Error) => void
    timer: NodeJS.Timeout
  }>
  lastUsed: number
}

export class PythonRuntimeManager {
  private workers = new Map<string, Worker>()
  private maxWorkers = 8
  
  constructor(
    private dataDir: string,
    private mcpAdapter: McpAdapter
  ) {}
  
  // 执行任务
  async execute(
    request: ExecutionRequest,
    signal: AbortSignal,
    timeoutMs = 60000
  ): Promise<ExecutionResponse> {
    signal.throwIfAborted()
    
    // 获取或创建 Worker
    let worker = this.workers.get(request.skillId)
    
    if (!worker) {
      worker = await this.createWorker(request)
      this.workers.set(request.skillId, worker)
    }
    
    worker.lastUsed = Date.now()
    
    // 发送请求
    const requestId = randomUUID()
    
    return new Promise<ExecutionResponse>((resolve, reject) => {
      const timer = setTimeout(() => {
        worker!.pending.delete(requestId)
        reject(new Error(`Execution timeout: ${request.skillId}.${request.tool}`))
      }, timeoutMs)
      
      const abort = () => {
        worker!.pending.delete(requestId)
        clearTimeout(timer)
        reject(new Error('Task cancelled'))
      }
      
      signal.addEventListener('abort', abort, { once: true })
      
      worker!.pending.set(requestId, { resolve, reject, timer })
      
      // JSON-RPC 请求
      const message = JSON.stringify({
        id: requestId,
        tool: request.tool,
        args: request.args,
        workspace: request.workspace
      }) + '\n'
      
      worker!.process.stdin.write(message, 'utf8', error => {
        if (error) {
          worker!.pending.delete(requestId)
          clearTimeout(timer)
          reject(new Error(`Failed to send request: ${error.message}`))
        }
      })
    })
  }
  
  // 创建 Worker 进程
  private async createWorker(request: ExecutionRequest): Promise<Worker> {
    // 准备 venv
    let venvPath: string | null = null
    let pythonExecutable = 'python3'
    
    if (request.venv) {
      venvPath = path.join(this.dataDir, 'venvs', request.skillId)
      
      if (!fs.existsSync(venvPath)) {
        await this.createVenv(venvPath, request.skillPath)
      }
      
      pythonExecutable = path.join(
        venvPath,
        process.platform === 'win32' ? 'Scripts/python.exe' : 'bin/python'
      )
    }
    
    // 启动进程
    const entryPath = path.join(request.skillPath, 'index.py')
    
    const child = spawn(pythonExecutable, [entryPath, '--runtime-mode'], {
      cwd: request.skillPath,
      env: {
        ...process.env,
        SKILL_ID: request.skillId,
        PYTHONIOENCODING: 'utf-8',
        PYTHONUTF8: '1',
        VIRTUAL_ENV: venvPath || undefined
      },
      stdio: ['pipe', 'pipe', 'pipe'],
      windowsHide: true
    })
    
    const worker: Worker = {
      skillId: request.skillId,
      process: child,
      venvPath,
      pending: new Map(),
      lastUsed: Date.now()
    }
    
    // 处理 stdout（JSON-RPC 响应）
    let buffer = ''
    child.stdout.on('data', (chunk: Buffer) => {
      buffer += chunk.toString('utf8')
      
      let newline = buffer.indexOf('\n')
      while (newline >= 0) {
        const line = buffer.slice(0, newline).trim()
        buffer = buffer.slice(newline + 1)
        newline = buffer.indexOf('\n')
        
        if (!line) continue
        
        try {
          const response = JSON.parse(line)
          const pending = worker.pending.get(response.id)
          
          if (pending) {
            clearTimeout(pending.timer)
            worker.pending.delete(response.id)
            
            if (response.error) {
              pending.reject(new Error(response.error))
            } else {
              pending.resolve(response)
            }
          }
        } catch (error) {
          console.error(`Failed to parse response:`, line, error)
        }
      }
    })
    
    // 处理 stderr
    child.stderr.on('data', (chunk: Buffer) => {
      console.error(`[Worker ${request.skillId}]`, chunk.toString('utf8'))
    })
    
    // 进程退出
    child.on('close', (code) => {
      console.log(`Worker exited: ${request.skillId}, code=${code}`)
      this.workers.delete(request.skillId)
      
      // 拒绝所有待处理请求
      for (const [id, pending] of worker.pending) {
        clearTimeout(pending.timer)
        pending.reject(new Error(`Worker exited: code=${code}`))
      }
      worker.pending.clear()
    })
    
    return worker
  }
  
  // 创建虚拟环境
  private async createVenv(venvPath: string, skillPath: string): Promise<void> {
    console.log(`Creating venv: ${venvPath}`)
    
    // python -m venv <path>
    await new Promise<void>((resolve, reject) => {
      const child = spawn('python3', ['-m', 'venv', venvPath], {
        stdio: 'inherit'
      })
      
      child.on('close', code => {
        if (code === 0) {
          resolve()
        } else {
          reject(new Error(`venv creation failed: code=${code}`))
        }
      })
    })
    
    // 安装依赖
    const requirementsPath = path.join(skillPath, 'requirements.txt')
    if (fs.existsSync(requirementsPath)) {
      const pipPath = path.join(
        venvPath,
        process.platform === 'win32' ? 'Scripts/pip.exe' : 'bin/pip'
      )
      
      await new Promise<void>((resolve, reject) => {
        const child = spawn(pipPath, ['install', '-r', requirementsPath], {
          stdio: 'inherit'
        })
        
        child.on('close', code => {
          if (code === 0) {
            resolve()
          } else {
            reject(new Error(`pip install failed: code=${code}`))
          }
        })
      })
    }
  }
  
  // 停止 Worker
  async stopWorker(skillId: string): Promise<void> {
    const worker = this.workers.get(skillId)
    if (!worker) return
    
    this.workers.delete(skillId)
    
    // 拒绝所有待处理请求
    for (const [id, pending] of worker.pending) {
      clearTimeout(pending.timer)
      pending.reject(new Error('Worker stopped'))
    }
    worker.pending.clear()
    
    // 终止进程
    worker.process.kill('SIGTERM')
    
    await new Promise<void>(resolve => {
      const timer = setTimeout(() => {
        worker.process.kill('SIGKILL')
        resolve()
      }, 3000)
      
      worker.process.once('close', () => {
        clearTimeout(timer)
        resolve()
      })
    })
  }
  
  // 停止所有 Workers
  async stopAll(): Promise<void> {
    await Promise.all(
      Array.from(this.workers.keys()).map(id => this.stopWorker(id))
    )
  }
  
  // 清理空闲 Workers（超过 5 分钟未使用）
  async cleanupIdle(): Promise<void> {
    const now = Date.now()
    const idleTimeout = 5 * 60 * 1000  // 5 分钟
    
    for (const [skillId, worker] of this.workers) {
      if (now - worker.lastUsed > idleTimeout) {
        await this.stopWorker(skillId)
      }
    }
  }
}
```

```python
"""MyPlane Skill SDK - Python"""
import sys
import json
import asyncio
from typing import Dict, Any, Callable, Optional
from abc import ABC, abstractmethod

class SkillBase(ABC):
    """Skill 基类"""
    
    def __init__(self):
        self.skill_id = os.getenv('SKILL_ID')
        self._tools: Dict[str, Callable] = {}
    
    def tool(self, name: str, description: str):
        """工具装饰器"""
        def decorator(func: Callable):
            self._tools[name] = func
            return func
        return decorator
    
    @abstractmethod
    async def setup(self):
        """初始化 Skill"""
        pass
    
    async def handle_call(self, tool: str, args: Dict[str, Any]) -> Any:
        """处理工具调用"""
        if tool not in self._tools:
            raise ValueError(f"Tool not found: {tool}")
        
        return await self._tools[tool](**args)
    
    async def run_runtime_mode(self):
        """Runtime 模式：持久运行，接收 JSON-RPC"""
        await self.setup()
        
        while True:
            try:
                line = sys.stdin.readline()
                if not line:
                    break
                
                request = json.loads(line.strip())
                request_id = request['id']
                
                try:
                    result = await self.handle_call(
                        request['tool'],
                        request['args']
                    )
                    
                    response = {
                        'id': request_id,
                        'output': result,
                        'elapsedMs': 0
                    }
                except Exception as e:
                    response = {
                        'id': request_id,
                        'error': str(e)
                    }
                
                print(json.dumps(response), flush=True)
            
            except Exception as e:
                sys.stderr.write(f"Runtime error: {e}\n")
                sys.stderr.flush()

def run_skill(skill_class):
    """运行 Skill"""
    if '--runtime-mode' in sys.argv:
        skill = skill_class()
        asyncio.run(skill.run_runtime_mode())
    else:
        print("Usage: python index.py --runtime-mode")
        sys.exit(1)
```

```python
"""MyPlane Storage SDK"""
import json

class SkillStorage:
    """Skill 数据存储"""
    
    def __init__(self, skill_id: str):
        self.skill_id = skill_id
        self.storage_path = f"/data/skills/{skill_id}/storage.json"
    
    def get(self, key: str, default: Any = None) -> Any:
        """读取数据"""
        try:
            with open(self.storage_path, 'r') as f:
                data = json.load(f)
            return data.get(key, default)
        except FileNotFoundError:
            return default
    
    def set(self, key: str, value: Any) -> None:
        """写入数据"""
        try:
            with open(self.storage_path, 'r') as f:
                data = json.load(f)
        except FileNotFoundError:
            data = {}
        
        data[key] = value
        
        with open(self.storage_path, 'w') as f:
            json.dump(data, f, indent=2)
```

- **Layer 1 (Agent Core)**: 智能决策
